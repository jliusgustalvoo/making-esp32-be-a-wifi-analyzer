// 2. SETUP (Pengaturan Awal)
// =================================================================
void setup() {
  Serial.begin(115200);
  WiFi.begin(ssid, password);
  
  Serial.print("Menghubungkan WiFi");
  while (WiFi.status() != WL_CONNECTED) { delay(500); Serial.print("."); }
  Serial.println("\nTerhubung! Buka IP: " + WiFi.localIP().toString());

  // POLISI LALU LINTAS 1: Jika browser meminta halaman utama ("/")
  server.on("/", []() {
    Serial.println("Browser meminta halaman HTML");
    server.send(200, "text/html", html_page); // Kirim paket Raksasa HTML
  });

  // POLISI LALU LINTAS 2: Jika tombol di klik, browser akan meminta ke ("/scan")
  server.on("/scan", []() {
    Serial.println("Browser meminta data WiFi. Mulai Scanning...");
    
    int n = WiFi.scanNetworks(); // ESP32 mendeteksi WiFi sekitar
    
    // Kita buat format JSON secara manual agar bisa dibaca JavaScript
    // Bentuk akhir JSON: [{"ssid":"Ayam Kate", "rssi":-35, "channel":4}]
    String json = "[";
    for (int i = 0; i < n; ++i) {
      if (i > 0) json += ","; // Tambah koma pemisah antar data
      json += "{\"ssid\":\"" + WiFi.SSID(i) + "\",";
      json += "\"rssi\":" + String(WiFi.RSSI(i)) + ",";
      json += "\"channel\":" + String(WiFi.channel(i)) + "}";
    }
    json += "]";

    // Kirim data JSON kembali ke browser
    server.send(200, "application/json", json);
    Serial.println("Data JSON berhasil dikirim!");
  });

  server.begin(); // Nyalakan server
}


// =================================================================

void loop() {
  // Biarkan library WebServer yang mengurus semua logika kedatangan browser
